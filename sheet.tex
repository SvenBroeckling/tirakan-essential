\newcommand{\sheettitle}[1]{%
        {\displayfont\fontsize{21pt}{24pt}\selectfont #1}
}

\newcommand{\panelheader}[1]{%
    \noindent\colorbox{ruleaccent}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{\color{white}\displayfont\large #1}}\par
}

\newcommand{\fieldwithheading}[2][1.2cm]{%
    \parbox[t]{\linewidth}{%
        \textbf{#2}\par
        \vspace{0.22cm}
        \rule{\linewidth}{0.4pt}\vspace{#1}%
    }%
}

\newcommand{\fieldwithcheckboxes}[3][0pt]{%
    \parbox[t]{\linewidth}{%
        \textbf{#2}\hfill\trackboxes{#3}\vspace{#1}%
    }%
}

\newcommand{\attributestepbox}[2]{%
    \tikz[baseline=-0.55ex]{
        \draw[draw=boxline, fill=white] (0,0) rectangle (2.05em,1.65em);
        \draw[draw=boxline!40] (0.16em,0.12em) -- (1.89em,1.53em);
        \node[anchor=north west, inner sep=0.16em] at (0,1.65em) {\fontsize{5pt}{5pt}\selectfont #1};
        \node[anchor=south east, inner sep=0.16em] at (2.05em,0) {\fontsize{4.8pt}{4.8pt}\selectfont #2};
    }%
}

\newcommand{\attributefield}[2][0pt]{%
    \parbox[t]{\linewidth}{%
        \textbf{#2}\hfill%
        \attributestepbox{0}{30}\hspace{0.18em}%
        \attributestepbox{1}{45}\hspace{0.18em}%
        \attributestepbox{2}{60}\hspace{0.18em}%
        \attributestepbox{3}{75}\hspace{0.18em}%
        \attributestepbox{4}{90}%
        \vspace{#1}%
    }%
}

\newcommand{\blankskillrow}[1][4]{%
    \parbox[t]{\linewidth}{%
        \rule{\dimexpr\linewidth-5.8em\relax}{0.4pt}\hfill\trackboxes{#1}%
    }%
}

\newcommand{\tinybox}[1]{%
    \tikz[baseline=(b.base)]\node[
        draw=boxline,
        fill=white,
        minimum width=1.0em,
        minimum height=1.0em,
        inner sep=0pt
    ] (b) {\strut #1};
}

\newcommand{\trackboxes}[1]{%
    \foreach \n in {1,...,#1}{%
        \tinybox{\textcolor{boxline!55}{\fontsize{5.5pt}{5.5pt}\selectfont \n}}%
    }%
}
