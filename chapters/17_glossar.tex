\section{Begriffe}\label{sec:begriffe}
\begin{description}[style=nextline,leftmargin=0pt]
\item[\textbf{Aktion}] Handlungseinheit im Konflikt; standardmäßig stehen pro Runde zwei Aktionen zur Verfügung.
\item[\textbf{Anrufungswert}] Summe aus Glaubensniveau und Ritus; bestimmt die Stärke einer Anrufung.
\item[\textbf{Arkana}] Die Ressource, die für das Zaubern verwendet wird.
\item[\textbf{Attribut}] Einer der sieben Grundwerte eines Charakters; fließt zusammen mit einer Fertigkeit in den Zielwert ein.
\item[\textbf{Abstammung}] Abstammungs-Prägung eines Charakters.
\item[\textbf{Ausrüstung}] Sammelbegriff für Waffen, Rüstungen, Schilde und Gebrauchsgegenstände eines Charakters.
\item[\textbf{Assistieren}] Aktion, die +10 auf eine unmittelbare Probe eines Verbündeten gibt.
\item[\textbf{Glaube}] Sammelbegriff für Bitten, Riten und Wunderwirkungen angerufener Mächte.
\item[\textbf{Bindung}] Eine lebendige Bindung an Person, Ort, Orden oder Pflicht.
\item[\textbf{Blutend}] Zustand, der am Ende jedes eigenen Zuges eine zusätzliche Wunde verursacht.
\item[\textbf{Segnung}] Zeremonielle Bitte zur Weihe, Reinigung oder Segnung von Personen, Orten oder Dingen.
\item[\textbf{Wendepunkt}] Zustand, wenn Burden exakt das Limit erreicht.
\item[\textbf{Durchatmen}] Kurze Erholung innerhalb einer Szene, die 1 Arkana, 1 Gunst oder 1 Bürde regeneriert.
\item[\textbf{Zerstört}] Zustand, in dem ausgewählte Ausrüstung unbrauchbar ist, bis sie repariert wird.
\item[\textbf{Bürde}] Wert für Stress, Erschöpfung, Schuld und Verderbnislast.
\item[\textbf{Brennend}] Zustand, bei dem Handlungen weitere Wunden verursachen, bis das Feuer gelöscht ist.
\item[\textbf{Weg}] Berufung oder sozialer Hintergrund eines Charakters.
\item[\textbf{Katastrophal}] Die höchste Eskalationsstufe eines riskant gewählten Zaubers und zugleich die schlimmste Misserfolgsmarge einer Probe bei 96--00.
\item[\textbf{Charakterbogen}] Regelblatt eines Charakters mit Werten, Prägungen, Ausrüstung und Werten.
\item[\textbf{Konsequenz}] Ergebnis einer Probe auf den Stufen Gering, Ernst oder Kritisch.
\item[\textbf{Verderbnis}] Langzeitfolge durch verbotene übernatürliche Praxis.
\item[\textbf{Deckung}] Schutz gegen Fernangriffe; leichte oder starke Deckung erhöht die Verteidigung.
\item[\textbf{Distanzsegment}] Relative Entfernungskategorie im Konflikt: Gebunden, Nah, Fern oder Weit entfernt.
\item[\textbf{Auszeit}] Phase zwischen Abenteuern, in der Charaktere mit begrenzten Aktionen heilen, forschen oder Kontakte pflegen.
\item[\textbf{Sterbend}] Zustand oberhalb der Wundschwelle; ohne Hilfe tödlich.
\item[\textbf{Gebunden, Nah, Fern, Weit entfernt}] Die vier relativen Distanzsegmente im Konflikt.
\item[\textbf{Erweiterte Prägung}] Eine zusätzliche Prägung, welche durch Wachstum erhalten werden kann.
\item[\textbf{Fertigkeit}] Erlernter Regelwert eines Charakters; wird mit einem Attribut zum Zielwert kombiniert.
\item[\textbf{Feldverband}] Kurze Erstversorgung, die Blutung stoppt, aber keine Wunden heilt.
\item[\textbf{Feldreparatur}] Improvisierte Reparatur eines kaputten Gegenstands für den Rest der Szene.
\item[\textbf{Fokus}] Ein Gegenstand oder Lebewesen, durch das Begabte Magie wirken.
\item[\textbf{Fortschritt}] Oberbegriff für Wachstum, Auszeit und andere langfristige Entwicklung eines Charakters.
\item[\textbf{Glaubensniveau}] Epochenwert, der Omen-Maximum und Anrufungen mitbestimmt.
\item[\textbf{Gabe}] Attribut für angeborene magische Begabung und das Wirken von Zaubern.
\item[\textbf{Schnellzauber}] Nicht-rituelles Zaubern mit Kosten von 1 Aktion.
\item[\textbf{Vollständige Rast}] Sichere volle Ruhephase mit besserer Heilung und voller Regeneration von Arkana und Gunst.
\item[\textbf{Große Anrufung}] Große, aufwendige Anrufung für öffentliche Wunder, Schutz oder Strafgericht.
\item[\textbf{Gunst}] Ressource für Anrufungen und Glaubenswunder.
\item[\textbf{Wachstum}] Fortschrittsmarke für überstandene Kosten und Verwundbarkeit.
\item[\textbf{Initiative Zielwurf}] Zielwert für die Reihenfolge zu Konfliktbeginn.
\item[\textbf{Konflikt}] Jede geordnete Spannungsszene mit Distanz, Aktionen und Reaktionen; Kampf ist der häufigste Fall.
\item[\textbf{Geringe Bitte}] Mittlere Anrufung für konkrete Wunderbitten wie Schutz, Reinigung oder Führung.
\item[\textbf{Magieniveau}] Epochenwert, der Auswirkungen auf Zauber hat.
\item[\textbf{Mal}] Verdiente Prägung mit dauerhaftem Vorteil und Verwundbarkeit.
\item[\textbf{Wegfacette}] Einmal-pro-Sitzung-Vorteil, der an einen Weg gekoppelt ist.
\item[\textbf{Gering}] Niedrige Zauberstufe mit erleichtertem Zielwert und ohne Zusatzkosten.
\item[\textbf{NSC}] Nichtspielercharakter, Gegner oder Nebenrolle unter Kontrolle der Spielleiterin.
\item[\textbf{Verpflichtung}] Dauerhafte Pflicht oder Schuld, die die Spielleiterin einmal pro Sitzung einfordern kann.
\item[\textbf{Oberhand}] Kleiner Vorteil bei beidseitigem Scheitern einer Gegenprobe.
\item[\textbf{Omen}] Seltene Gruppenressource für Eingriffe in Schicksal und Timing.
\item[\textbf{Gelegenheitsangriff}] Freier Nahkampfangriff gegen jemanden, der ein Gebunden ohne Zurückziehen-Aktion verlässt.
\item[\textbf{Anrufung}] Regelverfahren für Glaubensbitten.
\item[\textbf{Unsichere Rast}] Unsichere Ruhephase mit nur begrenzter Regeneration.
\item[\textbf{Prägung}] Sammelbegriff für Abstammung, Weg, Bindung und Mal.
\item[\textbf{Probe}] Standardverfahren zur Auflösung unsicherer Handlungen; wird mit 1W100 gegen einen Zielwert geworfen.
\item[\textbf{Schnellreaktion}] Vorab-Reaktion vor der ersten Konfliktrunde.
\item[\textbf{Reaktion}] Abwehr- oder Antwortaktion, die aus gehaltenen Aktionen bezahlt wird.
\item[\textbf{Relikt}] Gegenstand mit gespeicherter arkener oder heiliger Kraft.
\item[\textbf{Festgesetzt}] Zustand, der Bewegung und Sprint unterbindet.
\item[\textbf{Ritual}] Langsamer, vorbereiteter Zauber mit Kreisen, Material oder Assistenz.
\item[\textbf{Ritualvorbereitung}] Auszeit-Aktion zur Vorbereitung eines künftigen Zaubers oder einer Anrufung.
\item[\textbf{Regenerationsritual}] Ein Ritual, mit welchem ein Begabter Arkana wiederherstellen kann.
\item[\textbf{Schaden}] Grundschaden einer Waffe oder eines Effekts.
\item[\textbf{Schild}] Reaktive Ausrüstung zur Verteidigung gegen Nah- oder Fernangriffe.
\item[\textbf{Schutz}] Rüstungswert, der eingehenden Schaden einmalig reduziert.
\item[\textbf{Schwäche}] Konkreter ausnutzbarer Nachteil eines Gegners oder Effekts.
\item[\textbf{Versiegelung}] Schutzwert einer Rüstung gegen Zustände wie Erschüttert oder Blutend.
\item[\textbf{Schwer}] Erhöhte Zauberstufe mit höherem Risiko und zusätzlichen Arkana-Kosten.
\item[\textbf{Erschüttert}] Zustand, der den nächsten körperlichen Test erschwert.
\item[\textbf{Stoßgebet}] Schnelle Notsituation-Anrufung mit unmittelbarer Wirkung innerhalb einer Szene.
\item[\textbf{Kurze Rast}] Kurze sichere Pause mit begrenzter Heilung.
\item[\textbf{Stille, Zeichen, Eingreifen, Offenbarung}] Die Antworten einer Anrufung.
\item[\textbf{Meisteraktion}] Charakteristische Sonderaktion eines Gegners oder Wesens.
\item[\textbf{Belastete Bindung}] Vorübergehend erschöpfte Bindung ohne Bonus.
\item[\textbf{Tabu}] Dauerhaft verbotener Akt, dessen Bruch göttliche oder moralische Folgen auslöst.
\item[\textbf{Tanium}] Eine höchst magische Substanz, ähnlich einem Stein. Instabiler Speicher für Arkana.
\item[\textbf{Trainieren}] Auszeit-Aktion zum Sammeln von Trainingsvermerken für spätere Attributsteigerungen.
\item[\textbf{Verbotene Praxis}] Technik, die Zauberkosten senkt, aber Verderbnis verursacht.
\item[\textbf{Versiegelung}] Schutzwert einer Rüstung gegen Zustände wie Erschüttert oder Blutend.
\item[\textbf{Wille}] Attribut für Disziplin, Ausdauer und Überzeugung.
\item[\textbf{Wundschwelle}] Grenze, ab der weitere Wunden einen Charakter in Sterbend bringen.
\item[\textbf{Wunden}] Wert für körperlichen Schaden.
\item[\textbf{Zielwert}] Wert, der bei einer Probe unterwürfelt werden muss, um Erfolg zu erzielen.
\end{description}
