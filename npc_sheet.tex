\newcommand{\sheettitle}[1]{%
    {\displayfont\fontsize{13pt}{15pt}\selectfont #1}
}

\newcommand{\cardheader}[1]{%
    \noindent\colorbox{ruleaccent}{%
        \parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
            \color{white}\displayfont\normalsize #1%
        }%
    }\par
}

\newcommand{\linefield}[2][0.16cm]{%
    \parbox[t]{\linewidth}{%
        \textbf{#2}\par
        \vspace{0.04cm}
        \rule{\linewidth}{0.4pt}\vspace{#1}%
    }%
}

\newcommand{\tinybox}[1]{%
    \tikz[baseline=(b.base)]\node[
        draw=boxline,
        fill=white,
        minimum width=1.0em,
        minimum height=1.0em,
        inner sep=0pt
    ] (b) {\strut #1};
}

\newcommand{\trackboxes}[1]{%
    \foreach \n in {1,...,#1}{%
        \tinybox{\textcolor{boxline!55}{\fontsize{5.2pt}{5.2pt}\selectfont \n}}%
    }%
}
